#1

לֹא אִישׁ דְּבָרִים — סוד שמו של החומש

  • הַגִּבּוֹר שֶׁל הַסִּפּוּר: מֹשֶׁה רַבֵּנוּ, רוֹעֵה צֹאן בְּמִדְיָן, שֶׁטָּעַן לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁאֵינוֹ יוֹדֵעַ לְדַבֵּר.
  • הַקּוֹנְפְלִיקְט: הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹלֵחַ אוֹתוֹ לְהוֹצִיא אֶת עַם יִשְׂרָאֵל מִמִּצְרַיִם וּלְהַנְהִיגוֹ, וּמֹשֶׁה בָּטוּחַ שֶׁהוּא הָאָדָם הֲכִי לֹא מַתְאִים לַמְּשִׂימָה.
  • נְקֻדַּת הַשִּׂיא: אַרְבָּעִים שָׁנָה אַחַר כָּךְ, אוֹתוֹ מֹשֶׁה עוֹמֵד מוּל כָּל עַם יִשְׂרָאֵל וְנוֹשֵׂא, מִלָּה בְּמִלָּה, אֶת הַנְּאוּם הַגָּדוֹל בְּיוֹתֵר בְּתוֹלְדוֹת הָאֱנוֹשׁוּת — חוּמָשׁ שָׁלֵם.
  • מוּסַר הַשְּׂכֵל: כְּשֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא רוֹצֶה, אֵין דָּבָר כָּזֶה "אֲנִי לֹא יָכוֹל".

🔹 א. שְׁאֵלָה שֶׁמְּלַוָּה אוֹתָנוּ מֵהַשּׁוּרָה הָרִאשׁוֹנָה

בְּכָל שָׁנָה, כְּשֶׁאָנוּ פּוֹתְחִים חוּמָשׁ חָדָשׁ, יֵשׁ לוֹ שֵׁם שֶׁמְּסַפֵּר לָנוּ סִפּוּר. "בְּרֵאשִׁית" — כִּי שָׁם מַתְחִילָה בְּרִיאַת הָעוֹלָם. "שְׁמוֹת" — כִּי שָׁם נִמְנִים שְׁמוֹת בְּנֵי יִשְׂרָאֵל שֶׁיָּרְדוּ לְמִצְרַיִם. "בְּמִדְבָּר" — כִּי שָׁם מְסֻפָּרִים כָּל הַדְּבָרִים שֶׁקָּרוּ לָנוּ בַּמִּדְבָּר. וְהַיּוֹם אָנוּ פּוֹתְחִים חוּמָשׁ עִם שֵׁם מוּזָר בְּיוֹתֵר: "דְּבָרִים". פָּשׁוּט... דְּבָרִים. מָה זֶּה בִּכְלָל אוֹמֵר? נָכוֹן שֶׁזּוֹ הַמִּלָּה הָרִאשׁוֹנָה הַכְּתוּבָה בּוֹ — אֲבָל לְכָל חוּמָשׁ יֵשׁ שֵׁם עִם מַשְׁמָעוּת, וְ"דְּבָרִים" נִשְׁמָע כְּמוֹ הַשֵּׁם הֲכִי סְתָמִי שֶׁיֵּשׁ. בְּעֶזְרַת הַשֵּׁם, עַד סוֹף הַסִּפּוּר הַזֶּה תָּבִינוּ שֶׁזֶּהוּ אוּלַי הַשֵּׁם הֲכִי יָפֶה מִכֻּלָּם.

🔹 ב. "לֹא אִישׁ דְּבָרִים אָנֹכִי"

כְּדֵי לְהָבִין, צָרִיךְ לַחְזֹר אֲחוֹרָה, אֶל חוּמָשׁ שְׁמוֹת. שָׁם, לְיַד סְנֶה בּוֹעֵר בָּאֵשׁ בְּמִדְבָּר מִדְיָן, פָּגַשׁ מֹשֶׁה רַבֵּנוּ בַּפַּעַם הָרִאשׁוֹנָה אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא. וְהַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לוֹ מְשִׂימָה עֲצוּמָה: לֵךְ, הוֹצֵא אֶת עַמִּי מִמִּצְרַיִם, וֶהְיֵה הַמַּנְהִיג שֶׁלּוֹ.

וּמָה עָנָה מֹשֶׁה? הוּא לֹא קָפַץ בְּשִׂמְחָה. הוּא הִתְחִיל לְהִתְוַכֵּחַ. "אֲנִי? אֵיךְ אֲנִי אֵלֵךְ? אַתָּה יוֹדֵעַ שֶׁאֲנִי מְגַמְגֵּם? אֵיךְ בִּכְלָל אַצְלִיחַ לְדַבֵּר אֶל פַּרְעֹה, אֶל כָּל הָעָם?" וְאָז הוּא אָמַר אֶת הַמִּשְׁפָּט שֶׁכָּל הַסִּפּוּר הַזֶּה בָּנוּי עָלָיו:

"לֹא אִישׁ דְּבָרִים אָנֹכִי... כִּי כְבַד פֶּה וּכְבַד לָשׁוֹן אָנֹכִי" (שמות ד, י)

"אֲנִי לֹא אִישׁ שֶׁל דִּבּוּרִים," אָמַר מֹשֶׁה. "אֲנִי לֹא יוֹדֵעַ לְדַבֵּר יָפֶה. אֵיךְ אֲנִי יָכוֹל לִהְיוֹת הַמַּנְהִיג, הַנּוֹאֵם שֶׁל הָעָם?"

🔹 ג. "מִי שָׂם פֶּה לָאָדָם?"

וְאֵיךְ הֵגִיב הַקָּדוֹשׁ בָּרוּךְ הוּא לַטַּעֲנָה הַזֹּאת? הוּא לֹא רִחֵם עָלָיו וְחִפֵּשׂ שָׁלִיחַ אַחֵר. הוּא עָנָה לוֹ תְּשׁוּבָה שֶׁמְּנַעֶרֶת: "מִי שָׂם פֶּה לָאָדָם?" (שמות ד, יא) — מִי בִּכְלָל בָּרָא אֶת הַפֶּה שֶׁלְּךָ, אֶת הַלָּשׁוֹן שֶׁלְּךָ? עִם מִי אַתָּה מְדַבֵּר כָּרֶגַע? עִם בּוֹרֵא הָעוֹלָם. וְאִם בּוֹרֵא הָעוֹלָם הוּא זֶה שֶׁשּׁוֹלֵחַ אוֹתְךָ — בָּרוּר שֶׁתּוּכַל לְדַבֵּר. אֵין דָּבָר כָּזֶה "אֲנִי לֹא יָכוֹל", כְּשֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא רוֹצֶה. אִם הוּא רוֹצֶה — גַּם אַתָּה תֵּדַע לְדַבֵּר. קַל וָחֹמֶר שֶׁתּוּכַל לִהְיוֹת מַנְהִיג.

וּמֹשֶׁה רַבֵּנוּ הָלַךְ.

🔹 ד. הַנְּאוּם הַגָּדוֹל בְּתוֹלְדוֹת הָעוֹלָם

וְעַכְשָׁו בּוֹאוּ נִרְאֶה מָה קָרָה בַּסּוֹף לְאוֹתוֹ אֶחָד שֶׁאָמַר "לֹא אִישׁ דְּבָרִים אָנֹכִי". אַרְבָּעִים שָׁנָה אַחֲרֵי יְצִיאַת מִצְרַיִם, בְּעַרְבוֹת מוֹאָב, בְּאֶחָד בִּשְׁבָט, עוֹמֵד מֹשֶׁה רַבֵּנוּ מוּל כָּל עַם יִשְׂרָאֵל — מִלְיוֹנֵי אֲנָשִׁים — וּפוֹתֵחַ בְּדִבּוּר. וְהוּא לֹא מַפְסִיק. מֵאֶחָד בִּשְׁבָט וְעַד שִׁבְעָה בַּאֲדָר, חֲמִישָּׁה שָׁבוּעוֹת שְׁלֵמִים, הוּא נוֹשֵׂא אֶת הַנְּאוּם הַגָּדוֹל בְּיוֹתֵר בְּתוֹלְדוֹת הָאֱנוֹשׁוּת. נְאוּם שֶׁמְּסַכֵּם אֶת כָּל הַתּוֹרָה, אֶת כָּל הַהִיסְטוֹרְיָה שֶׁל הָעָם, וּמַדְרִיךְ אוֹתוֹ קָדִימָה — לְדוֹרֵי דוֹרוֹת, עַד הַיּוֹם הַזֶּה מַמָּשׁ. הַנְּאוּם הַזֶּה, מִלָּה בְּמִלָּה, נִכְתַּב וְהִשְׁתַּמֵּר. וְהוּא הַחוּמָשׁ שֶׁפּוֹתְחִים בּוֹ עַכְשָׁו.

וּמִי אָמַר אֶת הַנְּאוּם הָעֲנָק הַזֶּה? מִי עָמַד וְדִבֵּר בְּלִי הֶפְסֵק, שָׁבוּעַ אַחַר שָׁבוּעַ, מוּל עַם שָׁלֵם? אוֹתוֹ אֶחָד בְּדִיּוּק שֶׁפַּעַם אָמַר "לֹא אִישׁ דְּבָרִים אָנֹכִי" — "אֲנִי לֹא יוֹדֵעַ לְדַבֵּר".

וְלָכֵן קוֹרְאִים לַחוּמָשׁ הַזֶּה "דְּבָרִים". כִּי הוּא נֶאֱמַר עַל יְדֵי מִי שֶׁחָשַׁב שֶׁהוּא הֲכִי פָּחוֹת מְסֻגָּל לְדַבֵּר — וְנֶהְפַּךְ לַהֲכִי גָּדוֹל שֶׁדִּבֵּר אֵי פַּעַם. "אֵלֶּה הַדְּבָרִים אֲשֶׁר דִּבֶּר מֹשֶׁה אֶל כָּל יִשְׂרָאֵל" (דברים א, א) — הַדְּבָרִים שֶׁל מִי שֶׁטָּעַן שֶׁאֵין לוֹ דְּבָרִים בִּכְלָל.

💎 חִבּוּר עֶרְכִּי: ME - WE - YOU

  • ME (מֹשֶׁה רַבֵּנוּ): חָשַׁב עַל עַצְמוֹ שֶׁהוּא הֲכִי לֹא מַתְאִים לְדַבֵּר — וְדַוְקָא הוּא נֶהְפַּךְ לַנּוֹאֵם הַגָּדוֹל בְּיוֹתֵר בְּתוֹלְדוֹת הָעוֹלָם, כִּי הַקָּדוֹשׁ בָּרוּךְ הוּא רָצָה בְּכָךְ.
  • WE (אֲנַחְנוּ בַּכִּתָּה): כֻּלָּנוּ, מִדֵּי פַּעַם, חוֹשְׁבִים עַל עַצְמֵנוּ "אֲנִי לֹא מַצְלִיחַ בָּזֶה", "זֶה לֹא בִּשְׁבִילִי". וְלִפְעָמִים אֲנַחְנוּ אֲפִלּוּ לֹא מְנַסִּים, כִּי אָנוּ בְּטוּחִים מֵרֹאשׁ שֶׁלֹּא נַצְלִיחַ.
  • YOU (אַתָּה, כְּבָר הַיּוֹם): בַּפַּעַם הַבָּאָה שֶׁתַּחְשֹׁב "אֲנִי לֹא יָכוֹל" — תִּזְכֹּר אֶת מֹשֶׁה רַבֵּנוּ. תִּתְפַּלֵּל לַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁיַּעֲזֹר לְךָ, וְתִנְסֶה בְּכָל זֹאת. גַּם אַתָּה יָכוֹל לְהַפְתִּיעַ אֶת עַצְמְךָ.

📚 מְקוֹרוֹת

  • שְׁמוֹת ד, י–יב — טַעֲנָתוֹ שֶׁל מֹשֶׁה "לֹא אִישׁ דְּבָרִים אָנֹכִי" וּתְשׁוּבַת הַקָּדוֹשׁ בָּרוּךְ הוּא "מִי שָׂם פֶּה לָאָדָם".
  • דְּבָרִים א, א–ג — פְּתִיחַת הַחוּמָשׁ: "אֵלֶּה הַדְּבָרִים אֲשֶׁר דִּבֶּר מֹשֶׁה אֶל כָּל יִשְׂרָאֵל", וְצִיּוּן הַתַּאֲרִיךְ שֶׁבּוֹ הֵחֵל מֹשֶׁה לְבָאֵר אֶת הַתּוֹרָה.
