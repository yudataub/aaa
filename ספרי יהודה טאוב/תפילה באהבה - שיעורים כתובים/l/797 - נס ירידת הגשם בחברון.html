נס ירידת הגשם בחברון

797 נס ירידת הגשם בחברון בלה רז חלק א שלום ילדים יקרים! היום אספר לכם את הסיפור 'נס ירידת הגשם', שמסופר בספר 'שערי ירושלים', וכתב אותו רישר משה בן מנחם מנדל בשנת 1868. (אפשר לשים כל בית בכיתוב יפה בצד - יכולה לעזור בעיצוב אם צריך - מרגלית) לִפְנֵי שָׁנִים רַבּוֹת, בְּחֶבְרוֹן, כָּךְ סִפְּרוּ: הָיְתָה שְׁנַת בַּצֹּרֶת וּגְשָׁמִים לֹא יָרְדוּ. כָּל הַשָּׁנָה לֹא יָרַד מָטָר כְּלָל, סָבְלוּ תּוֹשָׁבֵי הָאָרֶץ, כִּי כְּלוּם לֹא גָּדַל.

לֹא הָיָה לֶחֶם, לֹא פֵּרוֹת, לֹא יְרָקוֹת, גַּם לֹא עֵשֶׂב לְמַאֲכָל לַבְּהֵמוֹת. כֻּלָּם רָעֲבוּ וְסָבְלוּ עַד מְאוֹד, גַּם הַתְיַבְּשׁוּ מֵי הַשְּׁתִיָּה בַּבּוֹרוֹת. וְאִם נִנְעֲלוּ שַׁעֲרֵי שָׁמַיִם לְבַקָּשׁוֹת, הַאִם לֹא יִפָּתְחוּ לִדְמָעוֹת וְלִתְפִלּוֹת? הַמַּצָּב לֹא יָכוֹל כָּךְ לְהַמְשִׁיךְ וְלִהְיוֹת. הַאִם יֵשׁ דְּבַר מָה שֶׁאֶפְשָׁר לַעֲשׂוֹת? הַמּוֹשֵׁל הַמֻּסְלְמִי הֶחֱלִיט וְצִוָּה: 'עַל הַתּוֹשָׁבִים הַיְּהוּדִים בִּפְקֻדָּה!

עֲלֵיכֶם לְהוֹרִיד גְּשָׁמִים בִּתְפִלָּה, וְאִם לֹא, אֲטַפֵּל בָּכֶם תְּחִלָּה'. מָסְרוּ הַיְּהוּדִים לַמּוֹשֵׁל בַּקָּשָׁה: 'אִם נֵרֵד לְתוֹךְ מְעָרַת הַמַּכְפֵּלָה, נַשְׁמִיעַ קוֹל זְעָקָה לַאֲבוֹת הָאֻמָּה, אָז תִּשָּׁמַע וְתִתְקַבֵּל שָׁם הַתְּפִלָּה'. בָּחֲרוּ הַיְּהוּדִים עֲשָׂרָה תַּלְמִידֵי חֲכָמִים, שֶׁיֵּרְדוּ חָמֵשׁ מַדְרֵגוֹת וְיִקְרְאוּ תְּהִלִּים. (תְּהִלִּים זֶהוּ סֵפֶר תְּפִלּוֹת, מִזְמוֹרִים, שֶׁחִבֵּר הַמֶּלֶךְ דָּוִד מָשִׁיחַ אֱלֹקִים) עָמְדוּ תַּלְמִידֵי הַחֲכָמִים בִּתְפִלָּה, וּבִקְּשׁוּ לָדַעַת הַאִם תְּפִלָּתָם הִתְקַבְּלָה?

שָׁלִיחַ שָׁלְחוּ, וְהוּא חָזַר וְאָמַר: 'הַשֶּׁמֶשׁ זוֹרַחַת'. הוֹסִיפוּ לָרֶדֶת עוֹד לְמַטָּה בְּפַחַד. קָרְאוּ סֵפֶר שֵׁנִי שֶׁל תְּהִלִּים וְסִיְּמוּ, שׁוּב לִבְדֹּק - מֵהַשָּׁלִיחַ בִּקְּשׁוּ. רָץ הַשָּׁלִיחַ בְּבֶהָלָה, וְאָמַר: 'לֹא, לֹא הִשְׁתַּנָּה שׁוּם דָּבָר'. יָרְדוּ עוֹד חָמֵשׁ מַדְרֵגוֹת בְּאֵימָה, וְסִיְּמוּ סֵפֶר שְׁלִישִׁי מִתּוֹךְ כַּוָּנָה. שָׁלְחוּ לִבְדֹּק מַרְאֶה שָׁמַיִם, אוּלַי הִשְׁתַּנָּה הוּא בֵּינְתַיִם.

'יֵשׁ עֲנָנִים', קָרָא הַשָּׁלִיחַ, 'וְגֶשֶׁם הִתְחִיל לָרֶדֶת'. חָשְׁבוּ עֲשָׂרָה: 'הָבָה נְכַוֵּן כַּוָּנָה אַחֶרֶת. נְבַקֵּשׁ גְּאֻלָּה מִשִּׁעְבּוּד לְעַם יִשְׂרָאֵל כִּי שְׁעַת רָצוֹן הִיא לִפְנֵי הָא-ל'. אֲבָל עוֹד לִפְנֵי שֶׁהִתְחִילוּ בִּתְפִלָּה, לְפֶתַע נָשְׁבָה רוּחַ מִתּוֹךְ הַמְּעָרָה, רוּחַ סְעָרָה אֲיֻמָּה אַדִּירָה, הַחוּצָה אוֹתָם דָּחֲפָה בְּחָזְקָה. (לעורך: לחתוך את הפעם הראשונה שאומרת קבילו החכמים אין) קִבְּלוּ הַחֲכָמִים, אֵין עֲדַיִן רְשׁוּת, לְקַצֵּר אֶת מֶשֶׁךְ הַגָּלוּת.

וְהַיּוֹם בָּאִים אָנוּ לְקִבְרֵי הָאָבוֹת לְהִתְפַּלֵּל כִּבְנֵי הַמָּקוֹם, הַשֶּׁבַח לָאֵל. חוֹזְרִים אָנוּ לְצַפּוֹת וּלְבַקֵּשׁ וּלְיַחֵל שֶׁתָּבוֹא גְּאֻלָּה בִּמְהֵרָה לְעַם יִשְׂרָאֵל. חלק ב נס ירידת הגשם בחברון, בלה רז ילדים יקרים! גם למוסלמים היה ידוע כח התפילה של הבנים היהודים על קברי האבות ולכן הם לא הרשו לנו בכלל לרדת ולהתפלל, חוץ מפעמים נדירות מאוד, בשעת בצורת או כשהיתה צרה גדולה. וזו אחת הפעמים שבהם זה קרה.

אנחנו לומדים מכך עד כמה תפילה על קברי האבות הקדושים היא דבר לא פשוט, קדוש ומסוגל לישועות. ברוכים הבאים להתפלל על קברי האבות!
